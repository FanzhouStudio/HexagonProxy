cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(HexagonProxy VERSION 0.5.4 LANGUAGES CXX RC)
if(MSVC AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  # Detect the installed compiler's localized /showIncludes prefix, even when
  # an older CMake cache was configured with another VSLANG.
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /? OUTPUT_VARIABLE cl_help ERROR_VARIABLE cl_help_err ENCODING UTF-8)
  if("${cl_help}${cl_help_err}" MATCHES "编译器")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  endif()
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Svg Concurrent Test)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/build/native-toolchain/yaml-cpp/CMakeLists.txt")
  message(FATAL_ERROR "Run tools/setup_native_toolchain.mjs first; yaml-cpp source is required")
endif()
add_subdirectory(build/native-toolchain/yaml-cpp yaml-cpp EXCLUDE_FROM_ALL)
add_library(hexagon_native STATIC
  native/src/storage.cpp native/src/jobs.cpp native/src/proxy.cpp
  native/src/accounts.cpp native/src/linux_workspace.cpp native/src/main_window.cpp native/src/web_workspace.cpp)
target_include_directories(hexagon_native PUBLIC native/src)
target_include_directories(hexagon_native PRIVATE "${CMAKE_SOURCE_DIR}/build/native-toolchain/webview2/build/native/include")
target_link_libraries(hexagon_native PUBLIC Qt6::Widgets Qt6::Network Qt6::Svg Qt6::Concurrent yaml-cpp::yaml-cpp wininet advapi32 shell32 ole32 oleaut32 shlwapi version "${CMAKE_SOURCE_DIR}/build/native-toolchain/webview2/build/native/x64/WebView2LoaderStatic.lib")
target_compile_definitions(hexagon_native PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
if(MSVC)
  target_compile_options(hexagon_native PUBLIC /utf-8 /W4 /permissive-)
endif()
qt_add_executable(HexagonProxy WIN32 native/src/main.cpp native/app.rc)
target_link_libraries(HexagonProxy PRIVATE hexagon_native)
qt_add_resources(HexagonProxy brand PREFIX "/brand" BASE "${CMAKE_SOURCE_DIR}/assets" FILES assets/ui/axolotl_mark.svg assets/app_icon.png assets/ui/themes.json)
qt_add_resources(HexagonProxy workspace_brand PREFIX "/workspace" BASE "${CMAKE_SOURCE_DIR}/native/src" FILES native/src/workspace_branding.js)
add_executable(hexagon_native_tests native/tests/tests.cpp)
target_link_libraries(hexagon_native_tests PRIVATE hexagon_native Qt6::Test)
qt_add_resources(hexagon_native_tests test_brand PREFIX "/brand" BASE "${CMAKE_SOURCE_DIR}/assets" FILES assets/ui/axolotl_mark.svg assets/app_icon.png assets/ui/themes.json)
qt_add_resources(hexagon_native_tests test_workspace_brand PREFIX "/workspace" BASE "${CMAKE_SOURCE_DIR}/native/src" FILES native/src/workspace_branding.js)
enable_testing()
add_test(NAME native_contracts COMMAND hexagon_native_tests -platform offscreen)
set(HEXAGON_PAYLOAD_ZIP "" CACHE FILEPATH "Embedded native deployment ZIP")
if(EXISTS "${HEXAGON_PAYLOAD_ZIP}")
  file(WRITE "${CMAKE_BINARY_DIR}/payload.rc" "101 RCDATA \"${HEXAGON_PAYLOAD_ZIP}\"\n")
  add_executable(HexagonProxyBundle WIN32 native/bootstrap/bootstrap.cpp native/app.rc "${CMAKE_BINARY_DIR}/payload.rc")
  set_target_properties(HexagonProxyBundle PROPERTIES AUTOMOC OFF OUTPUT_NAME HexagonProxy RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build/native-bundle" MSVC_RUNTIME_LIBRARY MultiThreaded)
  target_compile_options(HexagonProxyBundle PRIVATE /utf-8)
  target_compile_definitions(HexagonProxyBundle PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(HexagonProxyBundle PRIVATE bcrypt shell32 ole32)
endif()
